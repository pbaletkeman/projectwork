### Day 4 — Hash Map + Linked List

#### Segment 1 — Concept Study - Day 4

- LRU cache internals
- Hash map + doubly linked list

class Node:
    constructor(key, value):
        this.key = key
        this.value = value
        this.prev = null
        this.next = null

class LRUCache:
    constructor(capacity):
        this.capacity = capacity
        this.map = empty hash map          // Key -> Node
        this.head = new Node(0, 0)         // Dummy head
        this.tail = new Node(0, 0)         // Dummy tail
        this.head.next = this.tail
        this.tail.prev = this.head

    // Helper: Remove a node from its current position in the linked list
    function removeNode(node):
        prevNode = node.prev
        nextNode = node.next
        prevNode.next = nextNode
        nextNode.prev = prevNode

    // Helper: Insert a node right after the dummy head (most recently used)
    function insertAtHead(node):
        nextNode = this.head.next
        this.head.next = node
        node.prev = this.head
        node.next = nextNode
        nextNode.prev = node

    // Get value by key
    function get(key):
        if key not in this.map:
            return -1

        node = this.map[key]
        this.removeNode(node)
        this.insertAtHead(node)  // Move to front as most recently used
        return node.value

    // Put or update key-value pair
    function put(key, value):
        if key in this.map:
            node = this.map[key]
            node.value = value
            this.removeNode(node)
            this.insertAtHead(node)
        else:
            if size(this.map) == this.capacity:
                // Evict least recently used node (node right before tail)
                lruNode = this.tail.prev
                this.removeNode(lruNode)
                delete this.map[lruNode.key]

            newNode = new Node(key, value)
            this.insertAtHead(newNode)
            this.map[key] = newNode
